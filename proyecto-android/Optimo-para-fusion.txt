    override fun onDestroy() {
        for (p in pestanas) {
            val w = p.web
            if (w != null) {
                contenedor.removeView(w)
                w.destroy()
                p.web = null
            }
        }
        super.onDestroy()
    }

    @Suppress("DEPRECATION")
    override fun onBackPressed() {
        if (vistaCompleta != null) {
            quitarPantallaCompleta()
            return
        }
        if (paginaCompleta) {
            val ahora = android.os.SystemClock.elapsedRealtime()
            if (ahora - ultimoAtras <= 2000L) {
                ultimoAtras = 0L
                salirPaginaCompleta()
            } else {
                ultimoAtras = ahora
                toast("Pulsa atrás otra vez para salir de pantalla completa")
            }
            return
        }
        if (barraBuscar.visibility == View.VISIBLE) {
            cerrarBusqueda()
            return
        }
        if (omnibox.hasFocus()) {
            omnibox.clearFocus()
            ocultarTeclado(omnibox)
            return
        }
        val p = actual
        val w = p?.web
        if (w != null && w.canGoBack()) {
            w.goBack()
            return
        }
        if (p != null && pestanas.size > 1) {
            cerrarPestana(p)
            return
        }
        super.onBackPressed()
    }

    @Suppress("DEPRECATION")
    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)
        if (requestCode == REQ_ARCHIVO) {
            val cb = selectorArchivos
            selectorArchivos = null
            cb?.onReceiveValue(WebChromeClient.FileChooserParams.parseResult(resultCode, data))
        }
    }

    override fun onRequestPermissionsResult(
        requestCode: Int,
        permissions: Array<out String>,
        grantResults: IntArray
    ) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults)
        var todos = grantResults.isNotEmpty()
        for (r in grantResults) {
            if (r != PackageManager.PERMISSION_GRANTED) todos = false
        }
        when (requestCode) {
            REQ_PERMISOS_WEB -> {
                val par = pendientePermisosWeb
                pendientePermisosWeb = null
                if (par != null) {
                    if (todos) par.first.grant(par.second) else par.first.deny()
                }
            }
            REQ_UBICACION -> {
                val par = pendienteUbicacion
                pendienteUbicacion = null
                if (par != null) par.first.invoke(par.second, todos, false)
            }
            REQ_ALMACENAMIENTO -> {
                val tarea = pendienteDescarga
                pendienteDescarga = null
                if (todos && tarea != null) tarea() else toast("Sin permiso para guardar archivos")
            }
        }
    }

    private fun actualizarUi() {
        val p = actual ?: return
        aplicarPantallaSegura(p.privada)
        if (!omnibox.hasFocus()) {
            omnibox.setText(if (p.inicio) "" else p.url)
        }
        omnibox.background = fondoRedondeado(if (p.privada) paleta.campoPrivado else paleta.campo, 20.dp())
        if (p.privada) {
            candado.text = "🕶"
            candado.setTextColor(paleta.texto)
        } else if (p.url.startsWith("https://")) {
            candado.text = "🔒"
            candado.setTextColor(paleta.seguro)
        } else if (p.url.startsWith("http://")) {
            candado.text = "⚠"
            candado.setTextColor(paleta.advertencia)
        } else {
            candado.text = "🔍"
            candado.setTextColor(paleta.tenue)
        }
        val w = p.web
        btnAtras.alpha = if (w != null && w.canGoBack()) 1f else 0.35f
        btnAdelante.alpha = if (w != null && w.canGoForward()) 1f else 0.35f
        val guardado = p.url.isNotEmpty() && marcadores.contiene(p.url)
        btnMarcador.text = if (guardado) "★" else "☆"
        btnMarcador.setTextColor(if (guardado) paleta.acento else paleta.texto)
        btnPestanas.text = pestanas.size.toString()
        actualizarProgreso()
    }

    private fun actualizarProgreso() {
        val p = actual ?: return
        progreso.progress = p.progreso
        progreso.visibility = if (p.progreso in 0..99) View.VISIBLE else View.INVISIBLE
        btnRecargar.text = if (p.progreso in 0..99) "✕" else "↻"
    }

    private fun ocultarTeclado(v: View) {
        val imm = getSystemService(Context.INPUT_METHOD_SERVICE) as InputMethodManager
        imm.hideSoftInputFromWindow(v.windowToken, 0)
    }

    private fun mostrarTeclado(v: View) {
        val imm = getSystemService(Context.INPUT_METHOD_SERVICE) as InputMethodManager
        imm.showSoftInput(v, InputMethodManager.SHOW_IMPLICIT)
    }

    /** Con una pestaña privada visible no se permiten capturas de pantalla ni miniatura en «recientes». */
    private fun aplicarPantallaSegura(privada: Boolean) {
        if (privada) {
            window.addFlags(WindowManager.LayoutParams.FLAG_SECURE)
        } else {
            window.clearFlags(WindowManager.LayoutParams.FLAG_SECURE)
        }
    }

    private fun hayPrivadas(): Boolean {
        for (q in pestanas) {
            if (q.privada) return true
        }
        return false
    }

    /** Los diálogos son ventanas aparte: si hay pestañas privadas abiertas (p. ej. el selector de pestañas) también son seguros. */
    private fun nuevoDialogo(): AlertDialog.Builder {
        val tema = if (paleta.oscuro) android.R.style.Theme_Material_Dialog_Alert
        else android.R.style.Theme_Material_Light_Dialog_Alert
        return object : AlertDialog.Builder(this@MainActivity, tema) {
            override fun create(): AlertDialog {
                val d = super.create()
                if (hayPrivadas()) d.window?.addFlags(WindowManager.LayoutParams.FLAG_SECURE)
                return d
            }
        }
    }

    private fun opciones(titulo: String?, items: List<Pair<String, () -> Unit>>) {
        val b = nuevoDialogo()
        if (titulo != null) b.setTitle(titulo)
        val etiquetas = Array(items.size) { i -> items[i].first }
        b.setItems(etiquetas) { _, i -> items[i].second() }
        b.show()
    }

    private fun calcularUaEscritorio(): String {
        val base = WebSettings.getDefaultUserAgent(this)
        return base.replace(Regex("\\(Linux; Android[^)]*\\)"), "(X11; Linux x86_64)").replace(" Mobile", "")
    }

    @SuppressLint("SetJavaScriptEnabled")
    private fun webDe(p: Pestana): WebView {
        val existente = p.web
        if (existente != null) return existente
        val w = WebView(this)
        // Debe ir antes que cualquier otra llamada al WebView: la pestaña privada usa su propio perfil aislado.
        if (p.privada && perfilesOk) WebViewCompat.setProfile(w, PERFIL_PRIVADO)
        w.setBackgroundColor(paleta.fondo)
        val s = w.settings
        s.javaScriptEnabled = true
        s.domStorageEnabled = true
        s.databaseEnabled = true
        s.setSupportZoom(true)
        s.builtInZoomControls = true
        s.setDisplayZoomControls(false)
        s.useWideViewPort = true
        s.loadWithOverviewMode = true
        s.setSupportMultipleWindows(true)
        s.javaScriptCanOpenWindowsAutomatically = false
        s.allowFileAccess = false
        s.allowContentAccess = false
        s.setGeolocationEnabled(true)
        s.safeBrowsingEnabled = true
        s.mixedContentMode = WebSettings.MIXED_CONTENT_NEVER_ALLOW
        s.cacheMode = if (p.privada) WebSettings.LOAD_NO_CACHE else WebSettings.LOAD_DEFAULT
        if (p.escritorio) s.userAgentString = uaEscritorio
        w.setWebViewClient(ClienteWeb(p))
        w.setWebChromeClient(ClienteChrome(p))
        w.setDownloadListener { url, ua, disposicion, mime, _ ->
            descargar(url, ua, disposicion, mime, p.privada)
        }
        w.setFindListener { ordinal, total, _ ->
            contadorBuscar.text = if (total == 0) "0/0" else "${ordinal + 1}/$total"
        }
        w.setOnTouchListener { v, ev -> alTocar(v as WebView, p, ev) }
        w.setOnLongClickListener { v -> menuContextual(v as WebView, p) }
        aplicarAjustesWeb(w)
        anuncios.instalar(w)
        p.web = w
        return w
    }

    @Suppress("DEPRECATION")
    private fun aplicarAjustesWeb(w: WebView) {
        val oscuro = ajustes.oscurecerPaginas
        if (Build.VERSION.SDK_INT >= 33) {
            w.settings.isAlgorithmicDarkeningAllowed = oscuro
        } else if (Build.VERSION.SDK_INT >= 29) {
            w.settings.forceDark = if (oscuro) WebSettings.FORCE_DARK_ON else WebSettings.FORCE_DARK_OFF
        }
        gestorCookies(w).setAcceptThirdPartyCookies(w, ajustes.cookiesTerceros)
    }

    /** Gestor de cookies del perfil al que pertenece el WebView (el privado tiene el suyo). */
    private fun gestorCookies(w: WebView): CookieManager {
        if (perfilesOk) {
            try {
                return WebViewCompat.getProfile(w).cookieManager
            } catch (e: Throwable) {
                // Se usa el gestor general.
            }
        }
        return CookieManager.getInstance()
    }

    private fun gestorCookiesPrivado(): CookieManager? {
        if (!perfilesOk) return null
        return try {
            ProfileStore.getInstance().getOrCreateProfile(PERFIL_PRIVADO).cookieManager
        } catch (e: Throwable) {
            null
        }
    }

    /** Borra cookies, almacenamiento y caché de las pestañas privadas (el perfil entero). No hace nada si sigue en uso. */
    private fun borrarPerfilPrivado() {
        if (!perfilesOk) return
        try {
            ProfileStore.getInstance().deleteProfile(PERFIL_PRIVADO)
        } catch (e: Throwable) {
            // En uso o inexistente: se reintenta al cerrar la última pestaña privada o al arrancar.
        }
    }

    private fun crearPestana(privada: Boolean, url: String?, cargada: Boolean): Pestana? {
        if (privada && !perfilesOk) {
            toast("Tu WebView no admite pestañas privadas aisladas. Actualiza «Android System WebView».")
            return null
        }
        if (pestanas.size >= MAX_PESTANAS) {
            toast("Has alcanzado el máximo de pestañas")
            return null
        }
        val p = Pestana(siguienteId++, privada)
        p.cargada = cargada
        p.pendiente = url
        p.inicio = (url == null && !cargada)
        pestanas.add(p)
        activar(p)
        return p
    }

    private fun nuevaPestana(url: String?, privada: Boolean = false): Pestana? =
        crearPestana(privada, url, false)

    private fun activar(p: Pestana) {
        val anterior = actual
        if (anterior != null && anterior !== p) {
            val wa = anterior.web
            if (wa != null) {
                contenedor.removeView(wa)
                wa.onPause()
            }
        }
        actual = p
        val w = webDe(p)
        if (w.parent == null) {
            contenedor.addView(
                w,
                FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT)
            )
        }
        w.onResume()
        if (!p.cargada) {
            p.cargada = true
            val pend = p.pendiente
            p.pendiente = null
            if (pend != null) {
                p.inicio = false
                p.url = pend
                cargarUrl(p, w, pend)
            } else {
                mostrarInicio(p)
            }
        }
        cerrarBusqueda()
        actualizarUi()
    }

    private fun cerrarPestana(p: Pestana) {
        val idx = pestanas.indexOf(p)
        if (idx < 0) return
        pestanas.removeAt(idx)
        val w = p.web
        if (w != null) {
            contenedor.removeView(w)
            w.stopLoading()
            w.destroy()
            p.web = null
        }
        if (p === actual) {
            actual = null
            if (pestanas.isEmpty()) {
                nuevaPestana(null)
            } else {
                activar(pestanas[minOf(idx, pestanas.size - 1)])
            }
        } else {
            actualizarUi()
        }
        if (p.privada) {
            var quedan = false
            for (q in pestanas) {
                if (q.privada) quedan = true
            }
            if (!quedan) {
                // Última pestaña privada: se destruye su perfil (cookies, almacenamiento y caché).
                borrarPerfilPrivado()
                Handler(Looper.getMainLooper()).postDelayed({
                    if (!hayPrivadas()) borrarPerfilPrivado()
                }, 1500L)
            }
        }
    }

    private fun cargarEn(p: Pestana, url: String) {
        p.cargada = true
        p.pendiente = null
        p.inicio = false
        p.url = url
        p.titulo = ""
        cargarUrl(p, webDe(p), url)
        if (p === actual) actualizarUi()
    }

    /**
     * Punto único para cargar una URL. Un sitio por http (salvo la red local) nunca se abre sin más:
     * con «HTTPS primero» se intenta https y, si falla, se pregunta; sin él, se pregunta directamente.
     */
    private fun cargarUrl(p: Pestana, w: WebView, url: String, esEnlace: Boolean = false) {
        val sinCifrar = url.startsWith("http://", ignoreCase = true)
        val host = Omnibox.hostDe(url).lowercase()
        if (!sinCifrar || httpPermitido.contains(host) || Omnibox.esLocal(host)) {
            w.loadUrl(url)
            return
        }
        if (ajustes.httpsPrimero) {
            p.mejoradoDesde = url
            w.loadUrl("https://" + url.substring(7))
        } else {
            preguntarHttp(p, w, url, null, esEnlace)
        }
    }

    private fun preguntarHttp(p: Pestana, w: WebView, url: String, motivo: String?, esEnlace: Boolean) {
        val host = Omnibox.hostDe(url).lowercase()
        val aviso = StringBuilder()
        aviso.append(host).append(" no ofrece una conexión cifrada (HTTPS).")
        if (motivo != null) aviso.append(" (").append(motivo).append(")")
        aviso.append("\n\nSi continúas, lo que veas o envíes puede ser leído o modificado por terceros. No escribas contraseñas ni datos personales.")
        val volver: () -> Unit = {
            if (!esEnlace) mostrarBloqueoHttp(w, url, motivo)
        }
        nuevoDialogo()
            .setTitle("Conexión sin cifrar")
            .setMessage(aviso.toString())
            .setPositiveButton("Volver") { _, _ -> volver() }
            .setNegativeButton("Continuar sin cifrar") { _, _ ->
                httpPermitido.add(host)
                w.loadUrl(url)
            }
            .setOnCancelListener { volver() }
            .show()
    }

    private fun mostrarBloqueoHttp(w: WebView, url: String, motivo: String?) {
        val destino = if (motivo != null) "https://" + url.substring(7) else url
        val texto = motivo ?: "Conexión sin cifrar bloqueada."
        w.loadDataWithBaseURL(destino, Paginas.error(paleta.colores, destino, texto), "text/html", "utf-8", destino)
    }

    private fun mostrarInicio(p: Pestana) {
        p.cargada = true
        p.pendiente = null
        p.inicio = true
        p.url = ""
        p.titulo = "Nueva pestaña"
        p.progreso = 100
        webDe(p).loadDataWithBaseURL("optimo://inicio", paginaInicio(), "text/html", "utf-8", "optimo://inicio")
        if (p === actual) actualizarUi()
    }

    private fun paginaInicio(): String {
        val atajos = ArrayList<Pair<String, String>>()
        for (e in marcadores.todos()) {
            atajos.add(Pair(e.titulo, e.url))
            if (atajos.size >= 8) break
        }
        if (atajos.isEmpty()) atajos.addAll(Paginas.ATAJOS_POR_DEFECTO)
        return Paginas.inicio(paleta.colores, Omnibox.motor(ajustes.motor), atajos)
    }

    private fun fijarUrl(p: Pestana, url: String?) {
        if (url == null) return
        if (url.startsWith("optimo://") || ((url.startsWith("data:") || url == "about:blank") && p.inicio)) {
            p.inicio = true
            p.url = ""
        } else {
            p.inicio = false
            p.url = url
        }
    }

    private fun ir(texto: String) {
        val r = Omnibox.resolver(texto, Omnibox.motor(ajustes.motor))
        when (r) {
            is Omnibox.Resolucion.Vacio -> return
            is Omnibox.Resolucion.Navegar -> cargarEnActual(r.url)
            is Omnibox.Resolucion.Buscar -> cargarEnActual(r.url)
            is Omnibox.Resolucion.Externo -> abrirExterno(r.url)
            is Omnibox.Resolucion.NoSoportado -> toast("Esquema no soportado: " + r.esquema)
        }
        omnibox.clearFocus()
        ocultarTeclado(omnibox)
    }

    private fun cargarEnActual(url: String) {
        val p = actual
        if (p == null) {
            nuevaPestana(url)
        } else {
            cargarEn(p, url)
            p.web?.requestFocus()
        }
    }

    private fun recargarOParar() {
        val p = actual ?: return
        val w = p.web ?: return
        if (p.progreso in 0..99) {
            w.stopLoading()
            p.progreso = 100
            actualizarUi()
        } else if (p.inicio) {
            mostrarInicio(p)
        } else {
            w.reload()
        }
    }

    private fun alternarMarcador() {
        val p = actual ?: return
        if (p.inicio || p.url.isEmpty()) {
            toast("No hay nada que guardar")
            return
        }
        if (marcadores.contiene(p.url)) {
            marcadores.quitar(p.url)
            toast("Marcador eliminado")
        } else {
            marcadores.agregar(Entrada(tituloDe(p), p.url, System.currentTimeMillis()))
            toast("Marcador guardado")
        }
        actualizarUi()
    }

    private fun alternarEscritorio() {
        val p = actual ?: return
        p.escritorio = !p.escritorio
        val w = p.web ?: return
        w.settings.userAgentString = if (p.escritorio) uaEscritorio else null
        w.settings.useWideViewPort = true
        w.settings.loadWithOverviewMode = true
        if (p.inicio) mostrarInicio(p) else w.reload()
        toast(if (p.escritorio) "Versión de escritorio" else "Versión móvil")
    }

        barraBuscar.visibility = View.VISIBLE
        campoBuscar.setText("")
        contadorBuscar.text = ""

            ocultarTeclado(campoBuscar)
        }
    }

            items.add(Pair(t, a))

            ajustes.httpsPrimero = !ajustes.httpsPrimero
            mostrarAjustes()
        }
        add("Oscurecer páginas web: " + si(ajustes.oscurecerPaginas)) {
            ajustes.oscurecerPaginas = !ajustes.oscurecerPaginas
            for (q in pestanas) {
                val w = q.web
                if (w != null) aplicarAjustesWeb(w)
            }
            mostrarAjustes()
        }
        add("Permitir cookies de terceros: " + si(ajustes.cookiesTerceros)) {
            ajustes.cookiesTerceros = !ajustes.cookiesTerceros
            for (q in pestanas) {
                val w = q.web

            mostrarAjustes()

    private fun mostrarLista(titulo: String, lista: ListaPersistente, vacio: String, conBorrarTodo: Boolean) {
        if (lista.todos().isEmpty()) {
            toast(vacio)
            return
        }
        val vista = ListView(this)
        val adaptador = object : ArrayAdapter<Entrada>(
            this, android.R.layout.simple_list_item_2, android.R.id.text1, lista.todos()
        ) {
            override fun getView(position: Int, convertView: View?, parent: ViewGroup): View {
                val v = super.getView(position, convertView, parent)
                val e = getItem(position)
                val t1 = v.findViewById<TextView>(android.R.id.text1)
                val t2 = v.findViewById<TextView>(android.R.id.text2)
                if (e != null) {
                    t1.text = if (e.titulo.isNotEmpty()) e.titulo else e.url
                    t2.text = e.url
                }
                t1.maxLines = 1
                t2.maxLines = 1
                t1.ellipsize = TextUtils.TruncateAt.END
                t2.ellipsize = TextUtils.TruncateAt.END
                t1.setTextColor(paleta.texto)
                t2.setTextColor(paleta.tenue)
                return v
            }
        }
        vista.setAdapter(adaptador)
        var dialogo: AlertDialog? = null
        vista.setOnItemClickListener { _, _, pos, _ ->
            val e = lista.todos()[pos]
            dialogo?.dismiss()
            cargarEnActual(e.url)
        }
        vista.setOnItemLongClickListener { _, _, pos, _ ->
            val e = lista.todos()[pos]
            val items = ArrayList<Pair<String, () -> Unit>>()
            items.add(Pair("Abrir en pestaña nueva", {
                dialogo?.dismiss()
                nuevaPestana(e.url)
                Unit
            }))
            items.add(Pair("Eliminar", {
                lista.quitar(e.url)
                adaptador.notifyDataSetChanged()
                actualizarUi()
            }))
            opciones(null, items)
            true
        }
        val b = nuevoDialogo().setTitle(titulo).setView(vista).setNegativeButton("Cerrar", null)
        if (conBorrarTodo) {
            b.setNeutralButton("Borrar todo") { _, _ ->
                lista.vaciar()
                toast("Historial borrado")
            }
        }
        val d = b.create()
        dialogo = d
        d.show()
    }

    private fun mostrarPestanas() {
        val lista = ListView(this)
        var dialogo: AlertDialog? = null
        val adaptador = object : BaseAdapter() {
            override fun getCount(): Int = pestanas.size
            override fun getItem(position: Int): Any = pestanas[position]
            override fun getItemId(position: Int): Long = pestanas[position].id.toLong()
            override fun getView(position: Int, convertView: View?, parent: ViewGroup): View {
                val p = pestanas[position]
                val fila = LinearLayout(this@MainActivity)
                fila.setOrientation(LinearLayout.HORIZONTAL)
                fila.setGravity(Gravity.CENTER_VERTICAL)
                fila.setPadding(16.dp(), 8.dp(), 4.dp(), 8.dp())
                val textos = LinearLayout(this@MainActivity)

    private fun menuContextual(v: WebView, p: Pestana): Boolean {
        val r = v.hitTestResult
        val extra = r.extra
        when (r.type) {
            WebView.HitTestResult.SRC_ANCHOR_TYPE -> {
                if (extra == null) return false
                menuEnlace(extra, null, p.privada, v, p)
                return true
            }
            WebView.HitTestResult.SRC_IMAGE_ANCHOR_TYPE -> {
                val msg = Handler(Looper.getMainLooper()) { m ->
                    val enlace = m.data.getString("url")
                    if (enlace != null) menuEnlace(enlace, extra, p.privada, v, p)
                    true
                }.obtainMessage()
                v.requestFocusNodeHref(msg)
                return true
            }
            WebView.HitTestResult.IMAGE_TYPE -> {
                if (extra == null) return false
                menuEnlace(null, extra, p.privada, v, p)
                return true
            }
        }
        return false
    }

    private fun menuEnlace(enlace: String?, imagen: String?, privada: Boolean, web: WebView, p: Pestana) {
        val items = ArrayList<Pair<String, () -> Unit>>()
        fun add(t: String, a: () -> Unit) {
            items.add(Pair(t, a))
        }
        if (enlace != null) {
            add("Abrir en pestaña nueva") { nuevaPestana(enlace, privada) }
            add("Copiar enlace") { copiar(enlace) }
            add("Compartir enlace") { compartir(enlace) }
        }
        if (imagen != null) {
            add("Abrir imagen en pestaña nueva") { nuevaPestana(imagen, privada) }
            add("Descargar imagen") {
                descargar(imagen, actual?.web?.settings?.userAgentString, null, null, actual?.privada == true)
            }
        }
        add("Bloquear este anuncio…") { bloquearAnuncio(web, p, imagen, p.toqueX, p.toqueY) }
        opciones(enlace ?: imagen, items)
    }

    /** Registra la última pulsación y, en modo «elegir», convierte el siguiente toque en la selección. */
    private fun alTocar(v: WebView, p: Pestana, ev: android.view.MotionEvent): Boolean {
        if (ev.actionMasked == android.view.MotionEvent.ACTION_DOWN) {
            p.toqueX = ev.x
            p.toqueY = ev.y
        }
        if (!p.eligiendo) return false
        when (ev.actionMasked) {
            android.view.MotionEvent.ACTION_UP -> {
                p.eligiendo = false
                bloquearAnuncio(v, p, null, ev.x, ev.y)
                return true
            }
            android.view.MotionEvent.ACTION_MOVE -> {
                val holgura = android.view.ViewConfiguration.get(this).scaledTouchSlop
                if (Math.abs(ev.x - p.toqueX) > holgura || Math.abs(ev.y - p.toqueY) > holgura) {
                    p.eligiendo = false
                    toast("Selección cancelada")
                    return false
                }
                return true
            }
            android.view.MotionEvent.ACTION_CANCEL -> {
                p.eligiendo = false
                return false
            }
        }
        return true
    }

    /** Identifica el anuncio en (x, y) —píxeles de pantalla— y ofrece ocultarlo y/o bloquear su dominio. */
    @Suppress("DEPRECATION")
    private fun bloquearAnuncio(web: WebView, p: Pestana, imagen: String?, x: Float, y: Float) {
        val escala = if (web.scale > 0f) web.scale else resources.displayMetrics.density
        web.evaluateJavascript(anuncios.scriptElegir(x / escala, y / escala)) { crudo ->
            var datos: org.json.JSONObject? = null
            try {
                val v = org.json.JSONTokener(crudo ?: "null").nextValue()
                if (v is String) datos = org.json.JSONObject(v)
            } catch (e: Exception) {
                datos = null
            }
            if (datos == null) {
                toast("No he encontrado un anuncio ahí. Prueba «Bloquear un anuncio (tocar)» en el menú")
                return@evaluateJavascript
            }
            ofrecerReglas(web, p, imagen, datos)
        }
    }

    private fun ofrecerReglas(web: WebView, p: Pestana, imagen: String?, d: org.json.JSONObject) {
        val paginaHost = (try { Uri.parse(web.url ?: p.url).host } catch (e: Exception) { null } ?: "").lowercase()
        val sel = d.optString("sel")
        if (paginaHost.isEmpty() || sel.isEmpty()) {
            toast("No se pudo crear la regla para esta página")
            return
        }
        var dom = d.optString("dom").lowercase().removePrefix("www.")
        if (dom.isEmpty() && imagen != null) {
            dom = (try { Uri.parse(imagen).host } catch (e: Exception) { null } ?: "").lowercase().removePrefix("www.")
        }
        // Nunca se ofrece bloquear el propio dominio de la página (o uno emparentado): rompería el sitio entero.
        val propio = dom.isEmpty() || dom == paginaHost || paginaHost.endsWith(".$dom") || dom.endsWith(".$paginaHost") ||
            UtilDominio.normalizar(dom) == null
        val sitio = ReglasUsuario.clave(paginaHost)
        val n = d.optInt("n", 1)
        val etiquetas = ArrayList<String>()
        etiquetas.add("Ocultar en $sitio (" + d.optString("tag") + " " + d.optInt("w") + "×" + d.optInt("h") +
            (if (n > 1) ", $n elementos" else "") + ")\n" + sel)
        if (!propio) etiquetas.add("Bloquear el dominio $dom en todas partes")
        val marcadas = BooleanArray(etiquetas.size) { i -> i == 0 }
        nuevoDialogo()
            .setTitle("Bloquear este anuncio")
            .setMultiChoiceItems(etiquetas.toTypedArray(), marcadas) { _, i, v -> marcadas[i] = v }
            .setPositiveButton("Añadir") { _, _ ->
                var hecho = false
                if (marcadas[0] && ReglasUsuario.agregarOcultar(sitio, sel)) {
                    hecho = true
                    reinstalarAnuncios()
                    anuncios.aplicarAhora(web, paginaHost)
                }
                if (marcadas.size > 1 && marcadas[1] && ReglasUsuario.agregarDominio(dom)) hecho = true
                toast(if (hecho) "Regla añadida. Recarga la página para ver el efecto completo" else "No se añadió nada")
            }
            .setNegativeButton("Cancelar", null)
            .show()
    }

    private fun mostrarReglasUsuario() {
        val ocultar = ReglasUsuario.listaOcultar()
        val dominios = ReglasUsuario.listaDominios()
        if (ocultar.isEmpty() && dominios.isEmpty()) {
            toast("Aún no has añadido reglas")
            return
        }
        val etiquetas = ArrayList<String>()
        for (r in ocultar) etiquetas.add("Ocultar en " + r.host + ": " + r.sel)
        for (dd in dominios) etiquetas.add("Bloquear dominio: $dd")
        val marcadas = BooleanArray(etiquetas.size)
        nuevoDialogo()
            .setTitle("Reglas añadidas (marca para borrar)")
            .setMultiChoiceItems(etiquetas.toTypedArray(), marcadas) { _, i, v -> marcadas[i] = v }
            .setPositiveButton("Borrar marcadas") { _, _ ->
                var borradas = 0
                for (i in ocultar.indices) if (marcadas[i]) { ReglasUsuario.quitarOcultar(ocultar[i]); borradas++ }
                for (j in dominios.indices) if (marcadas[ocultar.size + j]) { ReglasUsuario.quitarDominio(dominios[j]); borradas++ }
                if (borradas > 0) {
                    reinstalarAnuncios()
                    toast("$borradas regla(s) borradas. Recarga las páginas abiertas")
                }
            }
            .setNegativeButton("Cerrar", null)
            .show()
    }

    private fun descargar(url: String, ua: String?, disposicion: String?, mime: String?, privada: Boolean = false) {
        if (!(url.startsWith("http://") || url.startsWith("https://"))) {
            toast("Este tipo de descarga no es compatible")
            return
        }
        if (url.startsWith("http://")) {
            val h = Omnibox.hostDe(url).lowercase()
            if (!httpPermitido.contains(h) && !Omnibox.esLocal(h)) {
                toast("Descarga sin cifrar bloqueada")
                return
            }
        }
        if (Build.VERSION.SDK_INT < 29 &&
            checkSelfPermission(Manifest.permission.WRITE_EXTERNAL_STORAGE) != PackageManager.PERMISSION_GRANTED
        ) {
            pendienteDescarga = { descargar(url, ua, disposicion, mime, privada) }
            requestPermissions(arrayOf(Manifest.permission.WRITE_EXTERNAL_STORAGE), REQ_ALMACENAMIENTO)
            return
        }
        try {
            val nombre = URLUtil.guessFileName(url, disposicion, mime)
            val pet = DownloadManager.Request(Uri.parse(url))
            if (!mime.isNullOrEmpty()) pet.setMimeType(mime)
            if (!ua.isNullOrEmpty()) pet.addRequestHeader("User-Agent", ua)
            // La descarga desde una pestaña privada usa las cookies del perfil privado, nunca las de las normales.
            val gestor = if (privada) gestorCookiesPrivado() else CookieManager.getInstance()
            val cookie = gestor?.getCookie(url)
            if (!cookie.isNullOrEmpty()) pet.addRequestHeader("Cookie", cookie)
            pet.setTitle(nombre)
            pet.setNotificationVisibility(DownloadManager.Request.VISIBILITY_VISIBLE_NOTIFY_COMPLETED)
            pet.setDestinationInExternalPublicDir(Environment.DIRECTORY_DOWNLOADS, nombre)
            val dm = getSystemService(Context.DOWNLOAD_SERVICE) as DownloadManager
            dm.enqueue(pet)
            toast("Descargando $nombre")
        } catch (e: Exception) {
            toast("No se pudo iniciar la descarga")
        }
    }

    private fun abrirExterno(url: String) {
        try {
            val intent = if (url.startsWith("intent:")) {
                Intent.parseUri(url, Intent.URI_INTENT_SCHEME)
            } else {
                Intent(Intent.ACTION_VIEW, Uri.parse(url))
            }
            val respaldo = intent.getStringExtra("browser_fallback_url")
            intent.addCategory(Intent.CATEGORY_BROWSABLE)
            intent.component = null
            intent.selector = null
            try {
                startActivity(intent)
            } catch (e: ActivityNotFoundException) {
                if (respaldo != null && (respaldo.startsWith("https://") || respaldo.startsWith("http://"))) {
                    cargarEnActual(respaldo)
                } else {
                    toast("No hay ninguna aplicación para abrir este enlace")
                }
            }
        } catch (e: Exception) {
            toast("No se pudo abrir el enlace")
        }
    }

    private fun procesarIntent(i: Intent?) {
        if (i == null) return
        if (i.action == Intent.ACTION_VIEW) {
            val d = i.data
            if (d != null) abrirDesdeFuera(d.toString())
        } else if (i.action == Intent.ACTION_SEND) {
            val t = i.getStringExtra(Intent.EXTRA_TEXT)
            if (t != null) abrirDesdeFuera(t)
        }
    }

    private fun abrirDesdeFuera(texto: String) {
        val r = Omnibox.resolver(texto, Omnibox.motor(ajustes.motor))
        val url: String? = when (r) {
            is Omnibox.Resolucion.Navegar -> r.url
            is Omnibox.Resolucion.Buscar -> r.url
            else -> null
        }
        if (url == null) return
        val p = actual
        if (p != null && p.inicio && pestanas.size == 1) {
            cargarEn(p, url)
        } else {
            nuevaPestana(url)
        }
    }

    // ---------------------------------------------------------------- sesion

    private fun guardarSesion() {
        val arr = JSONArray()
        var activa = 0
        var n = 0
        for (p in pestanas) {
            if (p.privada) continue
            if (p === actual) activa = n
            arr.put(if (p.inicio) "" else p.url)
            n++
        }
        val cifrada = AlmacenSeguro.cifrarTexto(arr.toString(), "sesion")
        val ed = getSharedPreferences("optimo_sesion", Context.MODE_PRIVATE).edit()
        ed.remove("urls") // formato antiguo, en texto plano
        if (cifrada != null) ed.putString("urls_c", cifrada) else ed.remove("urls_c")
        ed.putInt("activa", activa)
        ed.apply()
    }

    private fun restaurarSesion() {
        var restaurada = false
        try {
            val sp = getSharedPreferences("optimo_sesion", Context.MODE_PRIVATE)
            val cifrada = sp.getString("urls_c", null)
            val texto = if (cifrada != null) AlmacenSeguro.descifrarTexto(cifrada, "sesion") else sp.getString("urls", null)
            if (texto != null) {
                val arr = JSONArray(texto)
                val n = minOf(arr.length(), 10)
                for (i in 0 until n) {
                    val u = arr.getString(i)
                    val p = Pestana(siguienteId++, false)
                    if (u.isEmpty()) {
                        p.inicio = true
                    } else {
                        p.url = u
                        p.pendiente = u
                    }
                    pestanas.add(p)
                }
                if (pestanas.isNotEmpty()) {
                    val activa = sp.getInt("activa", 0)
                    activar(pestanas[if (activa in 0 until pestanas.size) activa else 0])
                    restaurada = true
                }
            }
        } catch (e: Exception) {
            restaurada = false
        }
        if (!restaurada) {
            pestanas.clear()
            nuevaPestana(null)
        }
    }

    // ---------------------------------------------------------------- clientes del WebView

    private inner class ClienteWeb(private val p: Pestana) : WebViewClient() {

        override fun shouldOverrideUrlLoading(view: WebView?, request: WebResourceRequest?): Boolean {
            if (view == null || request == null) return false
            val uri = request.url
            val esquema = uri.scheme?.lowercase() ?: return false
            if (esquema == "http" || esquema == "https") {
                val host = (uri.host ?: "").lowercase()
                if (esquema == "http" && request.isForMainFrame &&
                    !httpPermitido.contains(host) && !Omnibox.esLocal(host)
                ) {
                    cargarUrl(p, view, uri.toString(), true)
                    return true
                }
                return false
            }
            if (esquema == "about" || esquema == "blob" || esquema == "data" || esquema == "javascript") return false
            if (esquema == "file" || esquema == "content") return true
            if (request.hasGesture()) abrirExterno(uri.toString())
            return true
        }

        override fun shouldInterceptRequest(view: WebView?, request: WebResourceRequest?): WebResourceResponse? {
            if (request != null && ajustes.bloquearAnuncios && !request.isForMainFrame) {
                val respuesta = anuncios.interceptar(request)
                if (respuesta != null) {
                    p.bloqueados.incrementAndGet()
                    return respuesta
                }
            }
            return null
        }

        override fun onPageStarted(view: WebView?, url: String?, favicon: Bitmap?) {
            super.onPageStarted(view, url, favicon)
            if (view != null) anuncios.alEmpezar(view, url)
            p.progreso = 0
            p.bloqueados.set(0)
            fijarUrl(p, url)
            if (p === actual) actualizarUi()
        }

        override fun onPageFinished(view: WebView?, url: String?) {
            super.onPageFinished(view, url)
            p.progreso = 100
            p.mejoradoDesde = null
            fijarUrl(p, url)
            val titulo = view?.title ?: ""
            if (titulo.isNotEmpty() && !p.inicio) p.titulo = titulo
            if (!p.privada && !p.inicio && url != null && titulo != Paginas.TITULO_ERROR &&
                (url.startsWith("http://") || url.startsWith("https://"))
            ) {
                historial.agregar(Entrada(if (titulo.isNotEmpty()) titulo else url, url, System.currentTimeMillis()))
            }
            if (p === actual) actualizarUi()
        }

        override fun doUpdateVisitedHistory(view: WebView?, url: String?, isReload: Boolean) {
            super.doUpdateVisitedHistory(view, url, isReload)
            fijarUrl(p, url)
            if (p === actual) actualizarUi()
        }

        override fun onReceivedError(view: WebView?, request: WebResourceRequest?, error: WebResourceError?) {
            super.onReceivedError(view, request, error)
            if (view == null || request == null || !request.isForMainFrame) return
            val fallida = request.url.toString()
            val descripcion = error?.description?.toString() ?: "Error de red"
            if (descripcion.contains("ERR_ABORTED")) return
            val original = p.mejoradoDesde
            if (original != null && fallida.startsWith("https://") &&
                Omnibox.hostDe(original).equals(request.url.host ?: "", ignoreCase = true) &&
                !descripcion.contains("ERR_CERT") && !descripcion.contains("ERR_SSL")
            ) {
                // El sitio no responde por https: se ofrece http, pero solo si la persona lo confirma.
                p.mejoradoDesde = null
                preguntarHttp(p, view, original, descripcion, false)
                return
            }
            view.loadDataWithBaseURL(
                fallida,
                Paginas.error(paleta.colores, fallida, descripcion),
                "text/html",
                "utf-8",
                fallida
            )
        }

        override fun onReceivedSslError(view: WebView?, handler: SslErrorHandler?, error: SslError?) {
            if (handler == null) return
            val urlError = error?.url ?: ""
            val host = Omnibox.hostDe(urlError).lowercase()
            val esPrincipal = host.isNotEmpty() && host == Omnibox.hostDe(p.url).lowercase()
            val motivo = when (error?.primaryError) {
                SslError.SSL_EXPIRED -> "El certificado ha caducado."
                SslError.SSL_UNTRUSTED -> "El certificado no es de una entidad de confianza."
                SslError.SSL_IDMISMATCH -> "El certificado no corresponde a este sitio."
                SslError.SSL_NOTYETVALID -> "El certificado aún no es válido."
                else -> "El certificado no es válido."
            }
            // Un fallo de certificado jamás debe acabar degradado a http.
            if (esPrincipal) p.mejoradoDesde = null

            // Internet: nunca se continúa con un certificado inválido (sin excepción ni botón para saltarlo).
            if (!Omnibox.esLocal(host)) {
                handler.cancel()
                if (esPrincipal && view != null) {
                    view.post {
                        view.loadDataWithBaseURL(
                            urlError,
                            Paginas.error(paleta.colores, urlError, motivo + " La conexión se ha bloqueado por seguridad."),
                            "text/html",
                            "utf-8",
                            urlError
                        )
                    }
                }
                return
            }

            // Red local (router, NAS...): certificados autofirmados habituales; solo la página principal y tras confirmar.
            if (sslPermitido.contains(host)) {
                handler.proceed()
                return
            }
            if (!esPrincipal) {
                handler.cancel()
                return
            }
            nuevoDialogo()
                .setTitle("Certificado no válido")
                .setMessage(host + "\n\n" + motivo + "\n\nEs un equipo de tu red local. Continúa solo si lo reconoces.")
                .setPositiveButton("Volver") { _, _ -> handler.cancel() }
                .setNegativeButton("Continuar (equipo local)") { _, _ ->
                    sslPermitido.add(host)
                    handler.proceed()
                }
                .setOnCancelListener { handler.cancel() }
                .show()
        }

        override fun onRenderProcessGone(view: WebView?, detail: RenderProcessGoneDetail?): Boolean {
            // El proceso de render murio: se destruye el WebView y se recrea al reactivar la pestaña.
            if (view == null || view !== p.web) return true
            val url = p.url
            contenedor.removeView(view)
            view.destroy()
            p.web = null
            p.cargada = false
            p.pendiente = if (url.isEmpty()) null else url
            p.inicio = url.isEmpty()
            p.progreso = 100
            Handler(Looper.getMainLooper()).post {
                if (p === actual) activar(p)
            }
            return true
        }
    }

    private inner class ClienteChrome(private val p: Pestana) : WebChromeClient() {

        override fun onProgressChanged(view: WebView?, newProgress: Int) {
            p.progreso = newProgress
            if (p === actual) actualizarProgreso()
        }

        override fun onReceivedTitle(view: WebView?, title: String?) {
            if (!title.isNullOrEmpty() && !p.inicio) p.titulo = title
        }

        override fun onCreateWindow(
            view: WebView?,
            isDialog: Boolean,
            isUserGesture: Boolean,
            resultMsg: Message?
        ): Boolean {
            if (!isUserGesture || resultMsg == null) return false
            val nueva = crearPestana(p.privada, null, true) ?: return false
            val transporte = resultMsg.obj as WebView.WebViewTransport
            transporte.webView = webDe(nueva)
            resultMsg.sendToTarget()
            return true
        }

        override fun onCloseWindow(window: WebView?) {
            Handler(Looper.getMainLooper()).post {
                if (pestanas.contains(p)) cerrarPestana(p)
            }

/** Elemento de marcadores o historial. */
class Entrada(val titulo: String, val url: String, val tiempo: Long)

/**
 * Lista persistente en un archivo JSON (marcadores e historial). Lo mas reciente va primero,
 * sin duplicados por URL y con un maximo. Las escrituras frecuentes se agrupan (guardarPronto).
 */
class ListaPersistente(private val archivo: File, private val maximo: Int) {

    private val datos = ArrayList<Entrada>()
    private val handler = Handler(Looper.getMainLooper())
    private var pendiente = false
    private val tarea = Runnable {
        pendiente = false
        guardar()
    }

    init {
        cargar()
    }

    fun todos(): List<Entrada> = datos

    fun contiene(url: String): Boolean {
        for (e in datos) {
            if (e.url == url) return true
        }
        return false
    }

    fun agregar(e: Entrada) {
        quitarSinGuardar(e.url)
        datos.add(0, e)
        while (datos.size > maximo) {
            datos.removeAt(datos.size - 1)
        }
        guardarPronto()
    }

    fun quitar(url: String) {
        if (quitarSinGuardar(url)) guardarPronto()
    }

    fun vaciar() {
        datos.clear()
        guardarAhora()
    }

    fun guardarPronto() {
        if (!pendiente) {
            pendiente = true
            handler.postDelayed(tarea, 3000L)
        }
    }

    fun guardarAhora() {
        handler.removeCallbacks(tarea)
        pendiente = false
        guardar()
    }

    private fun quitarSinGuardar(url: String): Boolean {
        var quitado = false
        val it = datos.iterator()
        while (it.hasNext()) {
            if (it.next().url == url) {
                it.remove()
                quitado = true
            }
        }
        return quitado
    }

    private fun cargar() {
        try {
            val texto = AlmacenSeguro.leerTexto(archivo) ?: return
            val arr = JSONArray(texto)
            for (i in 0 until arr.length()) {
                val o = arr.getJSONObject(i)
                datos.add(Entrada(o.optString("t"), o.getString("u"), o.optLong("h")))
            }
        } catch (e: Exception) {
            datos.clear()
        }
    }

    private fun guardar() {
        try {
            val arr = JSONArray()
            for (e in datos) {
                val o = JSONObject()
                o.put("t", e.titulo)
                o.put("u", e.url)
                o.put("h", e.tiempo)
                arr.put(o)
            }
            AlmacenSeguro.escribirTexto(archivo, arr.toString())
        } catch (e: Exception) {
            // Sin espacio o sin permiso: se pierde solo esta escritura.
        }
    }
}

package com.optimo.navegador

import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import android.util.Base64
import java.io.File
import java.security.InvalidKeyException
import java.security.KeyStore
import java.security.UnrecoverableKeyException
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

/**
 * Cifrado de los datos locales de Optimo (marcadores, historial, sesion, reglas del usuario).
 *
 * Misma idea que La Boveda: AES-256-GCM con una clave que vive en el Android Keystore y nunca sale de el.
 * Cada dato lleva una etiqueta (nombre del archivo) como datos autenticados, asi que un archivo cifrado no
 * se puede intercambiar por otro. Formato del archivo: "OPT1" + IV (12 bytes) + texto cifrado con etiqueta GCM.
 *
 * Los archivos antiguos en texto plano se leen y se migran a cifrado al instante. Si el Keystore no esta
 * disponible NO se escribe en claro: se pierde la escritura (privacidad antes que persistencia).
 * Sin dependencias: solo APIs del sistema.
 */
object AlmacenSeguro {

    private const val ALIAS = "optimo_datos_v1"
    private const val TAM_IV = 12
    private const val TAM_ETIQUETA_BITS = 128
    private val MAGIA = byteArrayOf(0x4F, 0x50, 0x54, 0x31) // "OPT1"

    private fun clave(): SecretKey {
        val ks = KeyStore.getInstance("AndroidKeyStore")
        ks.load(null)
        val existente = ks.getKey(ALIAS, null)
        if (existente is SecretKey) return existente
        val gen = KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore")
        gen.init(
            KeyGenParameterSpec.Builder(ALIAS, KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT)
                .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
                .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
                .setKeySize(256)
                .build()
        )
        return gen.generateKey()
    }

    /** Solo cuando la clave es inservible (invalidada o irrecuperable): se descarta para crear una nueva. */
    private fun descartarClave() {
        try {
            val ks = KeyStore.getInstance("AndroidKeyStore")
            ks.load(null)
            ks.deleteEntry(ALIAS)
        } catch (e: Exception) {
            // Nada mas que hacer.
        }
    }

    fun esCifrado(datos: ByteArray): Boolean {
        if (datos.size < MAGIA.size) return false
        for (i in MAGIA.indices) {
            if (datos[i] != MAGIA[i]) return false
        }
        return true
    }

    private fun cifrarBytes(claro: ByteArray, etiqueta: String): ByteArray? {
        for (intento in 0..1) {
            try {
                val c = Cipher.getInstance("AES/GCM/NoPadding")
                c.init(Cipher.ENCRYPT_MODE, clave())
                c.updateAAD(etiqueta.toByteArray(Charsets.UTF_8))
                val iv = c.iv
                val cifrado = c.doFinal(claro)
                return MAGIA + iv + cifrado
            } catch (e: Exception) {
                if (intento == 0 && (e is InvalidKeyException || e is UnrecoverableKeyException)) {
                    descartarClave()
                } else {
                    return null
                }
            }
        }
        return null
    }

    private fun descifrarBytes(datos: ByteArray, etiqueta: String): ByteArray? {
        if (!esCifrado(datos) || datos.size < MAGIA.size + TAM_IV + 16) return null
        return try {
            val iv = datos.copyOfRange(MAGIA.size, MAGIA.size + TAM_IV)
            val cifrado = datos.copyOfRange(MAGIA.size + TAM_IV, datos.size)
            val c = Cipher.getInstance("AES/GCM/NoPadding")
            c.init(Cipher.DECRYPT_MODE, clave(), GCMParameterSpec(TAM_ETIQUETA_BITS, iv))
            c.updateAAD(etiqueta.toByteArray(Charsets.UTF_8))
            c.doFinal(cifrado)
        } catch (e: Exception) {
            null
        }
    }

    /** Texto cifrado y en Base64, para guardarlo en SharedPreferences. Null si el Keystore falla. */
    fun cifrarTexto(texto: String, etiqueta: String): String? {
        val b = cifrarBytes(texto.toByteArray(Charsets.UTF_8), etiqueta) ?: return null
        return Base64.encodeToString(b, Base64.NO_WRAP)
    }

    fun descifrarTexto(base64: String, etiqueta: String): String? {
        return try {
            val claro = descifrarBytes(Base64.decode(base64, Base64.NO_WRAP), etiqueta) ?: return null
            String(claro, Charsets.UTF_8)
        } catch (e: Exception) {
            null
        }
    }

    /** Lee un archivo cifrado (o uno antiguo en texto plano, que migra). Null si no existe o no se puede descifrar. */
    fun leerTexto(f: File, etiqueta: String = f.name): String? {
        try {
            if (!f.exists()) return null
            val datos = f.readBytes()
            if (esCifrado(datos)) {
                val claro = descifrarBytes(datos, etiqueta) ?: return null
                return String(claro, Charsets.UTF_8)
            }
            val texto = String(datos, Charsets.UTF_8)
            escribirTexto(f, texto, etiqueta)
            return texto
        } catch (e: Exception) {
            return null
        }
    }

    /** Escribe cifrado y de forma atomica (archivo temporal + renombrar). False si no se pudo. */
    fun escribirTexto(f: File, texto: String, etiqueta: String = f.name): Boolean {
        val cifrado = cifrarBytes(texto.toByteArray(Charsets.UTF_8), etiqueta) ?: return false
        return try {
            val tmp = File(f.path + ".tmp")
            tmp.writeBytes(cifrado)
            if (!tmp.renameTo(f)) {
                f.writeBytes(cifrado)
                tmp.delete()
            }
            true
        } catch (e: Exception) {
            false
        }
    }
}

object UtilDominio {
    fun parseIpv4(entrada: String): Int? {
        val partes = entrada.trim().split('.')
        if (partes.size != 4) return null
        var resultado = 0
        for (p in partes) {
            if (p.isEmpty() || p.length > 3) return null
            val n = p.toIntOrNull() ?: return null
            if (n < 0 || n > 255) return null
            resultado = (resultado shl 8) or n
        }
        return resultado
    }

    /** Dominio en minúsculas, o null si no es válido (menos de 3 caracteres, sin punto, caracteres raros o IP). */
    fun normalizar(entrada: String): String? {
        var t = entrada.trim().lowercase()
        if (t.startsWith("http://")) t = t.substring(7)
        if (t.startsWith("https://")) t = t.substring(8)
        val barra = t.indexOf('/')
        if (barra >= 0) t = t.substring(0, barra)
        val dosPuntos = t.indexOf(':')
        if (dosPuntos >= 0) t = t.substring(0, dosPuntos)
        t = t.trim('.')
        if (t.length < 3 || t.length > 253 || !t.contains('.')) return null
        for (c in t) {
            val valido = (c in 'a'..'z') || (c in '0'..'9') || c == '-' || c == '.' || c == '_'
            if (!valido) return null
        }
        if (parseIpv4(t) != null) return null
        return t
    }
}

/**
 * Lee una línea de una lista de bloqueo en cualquiera de estos formatos y devuelve el dominio, o null si no sirve:
 *  - hosts:     «0.0.0.0 ads.ejemplo.com»  (AdAway, StevenBlack, 1Hosts)
 *  - dominios:  «ads.ejemplo.com»          (OISD, Firebog, anudeepND)
 *  - Adblock:   «||ads.ejemplo.com^»       (EasyList, HaGeZi, AdGuard) y excepciones «@@||ok.ejemplo.com^»
 * Solo se pueden aplicar reglas de dominio entero: se descartan rutas, comodines, modificadores y reglas cosméticas.
 */
object ParserListas {

    class Entrada(val dominio: String, val permitido: Boolean)

    private val IGNORADOS = setOf(
        "localhost", "localhost.localdomain", "local", "broadcasthost", "ip6-localhost", "ip6-loopback",
        "ip6-localnet", "ip6-mcastprefix", "ip6-allnodes", "ip6-allrouters", "ip6-allhosts", "0.0.0.0"
    )

    private val MODIFICADORES_ACEPTADOS = setOf("", "|", "\$important", "\$all")

    fun procesarLinea(linea: String): Entrada? {
        var s = linea.trim()
        if (s.isEmpty()) return null
        val c0 = s[0]
        if (c0 == '#' || c0 == '!' || c0 == '[') return null
        if (s.contains("##") || s.contains("#@#") || s.contains("#?#") || s.contains("#\$#")) return null

        if (s.startsWith("||") || s.startsWith("@@||")) {
            val permitido = s.startsWith("@@")
            val resto = s.substring(if (permitido) 4 else 2)
            val fin = resto.indexOf('^')
            if (fin < 0) return null
            if (!MODIFICADORES_ACEPTADOS.contains(resto.substring(fin + 1))) return null
            return crear(resto.substring(0, fin), permitido)
        }

        // Formato hosts o lista de dominios: se corta el comentario final.
        val almohadilla = s.indexOf('#')
        if (almohadilla > 0) s = s.substring(0, almohadilla).trim()
        val partes = s.split(' ', '\t').filter { it.isNotEmpty() }
        if (partes.isEmpty()) return null
        val candidato: String
        if (partes.size == 1) {
            candidato = partes[0]
        } else {
            val primero = partes[0]
            val esIp = UtilDominio.parseIpv4(primero) != null || primero.contains(':')
            if (!esIp) return null
            candidato = partes[1]
        }
        return crear(candidato, false)
    }

    private fun crear(crudo: String, permitido: Boolean): Entrada? {
        var d = crudo
        if (d.startsWith("*.")) d = d.substring(2)
        if (d.startsWith(".")) d = d.substring(1)
        if (d.contains('/') || d.contains('*') || d.contains(':') || d.contains('?') || d.contains('|')) return null
        val limpio = UtilDominio.normalizar(d) ?: return null
        if (IGNORADOS.contains(limpio)) return null
        return Entrada(limpio, permitido)
    }
}

/** Hash de 64 bits de un dominio, calculado de DERECHA a IZQUIERDA para comprobar todos sus sufijos en una pasada. */
object HashDominio {
    const val SEMILLA = 1125899906842597L
    const val PRIMO = 1099511628211L

    fun completo(dominio: String): Long {
        var h = SEMILLA
        var i = dominio.length - 1
        while (i >= 0) {
            h = (h xor dominio[i].code.toLong()) * PRIMO
            i--
        }
        return h
    }
}

/**
 * Conjunto de hashes de 64 bits en una tabla abierta de longs: ~16 bytes por dominio, sin objetos por entrada.
 * 600 000 dominios ocupan unos 16 MB y una consulta tarda menos de un microsegundo. El 0 marca «hueco».
 */
class ConjuntoHash64 private constructor(capacidad: Int) {
    private var tabla = LongArray(capacidad)

    var tamano = 0
        private set

    constructor() : this(1024)

    fun agregar(valor: Long): Boolean {
        val h = if (valor == 0L) 1L else valor
        if ((tamano + 1) * 10 > tabla.size * 6) crecer()
        if (insertar(tabla, h)) {
            tamano++
            return true
        }
        return false
    }

    fun contiene(valor: Long): Boolean {
        val h = if (valor == 0L) 1L else valor
        val t = tabla
        val mascara = t.size - 1
        var i = ((h xor (h ushr 32)).toInt()) and mascara
        while (true) {
            val v = t[i]
            if (v == 0L) return false
            if (v == h) return true
            i = (i + 1) and mascara
        }
    }

    private fun insertar(t: LongArray, h: Long): Boolean {
        val mascara = t.size - 1
        var i = ((h xor (h ushr 32)).toInt()) and mascara
        while (true) {
            val v = t[i]
            if (v == 0L) {
                t[i] = h
                return true
            }
            if (v == h) return false
            i = (i + 1) and mascara
        }
    }

    private fun crecer() {
        val nueva = LongArray(tabla.size * 2)
        for (v in tabla) {
            if (v != 0L) insertar(nueva, v)
        }
        tabla = nueva
    }

    fun escribir(salida: OutputStream) {
        val cab = ByteBuffer.allocate(8)
        cab.putInt(tamano)
        cab.putInt(tabla.size)
        salida.write(cab.array())
        val bloque = ByteBuffer.allocate(BLOQUE * 8)
        var i = 0
        while (i < tabla.size) {
            val n = minOf(BLOQUE, tabla.size - i)
            bloque.clear()
            for (j in 0 until n) bloque.putLong(tabla[i + j])
            salida.write(bloque.array(), 0, n * 8)
            i += n
        }
    }

    companion object {
        private const val BLOQUE = 8192

        fun leer(entrada: InputStream): ConjuntoHash64 {
            val din = DataInputStream(entrada)
            val tam = din.readInt()
            val cap = din.readInt()
            if (cap < 16 || cap > (1 shl 26) || (cap and (cap - 1)) != 0 || tam < 0 || tam > cap) {
                throw IllegalStateException("caché de filtro no válida")
            }
            val c = ConjuntoHash64(cap)
            val bytes = ByteArray(BLOQUE * 8)
            var i = 0
            while (i < cap) {
                val n = minOf(BLOQUE, cap - i)
                din.readFully(bytes, 0, n * 8)
                val bb = ByteBuffer.wrap(bytes, 0, n * 8)
                for (j in 0 until n) c.tabla[i + j] = bb.getLong()
                i += n
            }
            c.tamano = tam
            return c
        }
    }
}

/** Dominios bloqueados y excepciones. Un dominio bloqueado incluye sus subdominios; una excepción gana siempre. */
class FiltroListas {

    class Tablas(val bloqueo: ConjuntoHash64, val permitido: ConjuntoHash64)

    @Volatile
    private var tablas: Tablas? = null

    val cargado: Boolean get() = tablas != null

    val totalDominios: Int get() = tablas?.bloqueo?.tamano ?: 0

    fun reemplazar(nuevas: Tablas) {
        tablas = nuevas
    }

    /** NO_LISTADO, BLOQUEADO o PERMITIDO (excepción de las listas). */
    fun consultar(nombre: String): Int {
        val t = tablas ?: return NO_LISTADO
        return evaluar(t, nombre.lowercase().trim('.'))
    }

    private fun evaluar(t: Tablas, nombre: String): Int {
        var h = HashDominio.SEMILLA
        var puntos = 0
        var bloq = false
        var perm = false
        var i = nombre.length - 1
        while (i >= 0) {
            val c = nombre[i]
            h = (h xor c.code.toLong()) * HashDominio.PRIMO
            if (c == '.') puntos++
            // [i..fin] es un sufijo completo (empieza en una etiqueta). Los sufijos sin punto (TLD) no se miran.
            if ((i == 0 || nombre[i - 1] == '.') && puntos > 0) {
                if (t.permitido.contiene(h)) perm = true
                if (t.bloqueo.contiene(h)) bloq = true
            }
            i--
        }
        if (perm) return PERMITIDO
        return if (bloq) BLOQUEADO else NO_LISTADO
    }

    fun guardarCache(archivo: File, t: Tablas) {
        val tmp = File(archivo.parentFile, archivo.name + ".tmp")
        tmp.outputStream().buffered().use { out ->
            out.write(MAGIA)
            t.bloqueo.escribir(out)
            t.permitido.escribir(out)
        }
        if (!tmp.renameTo(archivo)) {
            archivo.delete()
            tmp.renameTo(archivo)
        }
    }

    /** Caché compilada (mucho más rápida que reprocesar las listas); null si no existe o está dañada. */
    fun leerCache(archivo: File): Tablas? {
        if (!archivo.exists()) return null
        return try {
            archivo.inputStream().buffered().use { entrada ->
                val magia = ByteArray(MAGIA.size)
                DataInputStream(entrada).readFully(magia)
                if (!magia.contentEquals(MAGIA)) {
                    null
                } else {
                    val bloqueo = ConjuntoHash64.leer(entrada)
                    val permitido = ConjuntoHash64.leer(entrada)
                    Tablas(bloqueo, permitido)
                }
            }
        } catch (e: Exception) {
            null
        }
    }

    companion object {
        const val NO_LISTADO = 0
        const val BLOQUEADO = 1
        const val PERMITIDO = 2
        private val MAGIA = "ADB1".toByteArray(Charsets.US_ASCII)
    }
}

            for (l in lista) {

                } catch (e: Exception) {
                    err++
                    mensajes.add(l.nombre + ": " + (e.message ?: e.javaClass.simpleName))
                }
            }
            var dominios = filtro.totalDominios
            if (ok > 0 || forzarCompilar || !filtro.cargado) {
                synchronized(cerrojo) { dominios = compilar(c) }
            }
            if (lista.isEmpty() || err < lista.size) a.ultimaActualizacion = System.currentTimeMillis()
            return if (err == 0) "Listas al día · $dominios dominios"
            else "Actualizado con $err error(es): " + mensajes.joinToString("; ")
        } catch (e: OutOfMemoryError) {
            return "Memoria insuficiente: activa menos listas o evita las muy grandes"
        } catch (e: Exception) {
            return "Error: " + (e.message ?: e.javaClass.simpleName)
        } finally {
            enCurso.set(false)
        }
    }

    /** Devuelve true si la lista cambió, false si el servidor dijo «sin cambios». Lanza excepción si falla. */
    private fun descargar(c: Context, a: Ajustes, lista: ListaDef): Boolean {
        if (!lista.url.startsWith("https://")) throw IllegalArgumentException("solo se admiten URL https")
        val destino = archivoLista(c, lista.id)
        val conn = URL(lista.url).openConnection() as HttpURLConnection
        try {
            conn.connectTimeout = 15000
            conn.readTimeout = 30000
            conn.instanceFollowRedirects = true
            conn.setRequestProperty("User-Agent", "Optimo/1.2")
            if (destino.exists()) {
                val etag = a.etag(lista.id)
                val mod = a.modificado(lista.id)
                if (etag != null) conn.setRequestProperty("If-None-Match", etag)
                if (mod != null) conn.setRequestProperty("If-Modified-Since", mod)
            }
            val codigo = conn.responseCode
            if (codigo == HttpURLConnection.HTTP_NOT_MODIFIED) return false
            if (codigo != HttpURLConnection.HTTP_OK) throw IllegalStateException("el servidor respondió $codigo")

            val tmp = File(destino.parentFile, lista.id + ".descarga")
            var bytes = 0L
            conn.inputStream.use { entrada ->
                tmp.outputStream().buffered().use { salida ->
                    val buf = ByteArray(32768)
                    while (true) {
                        val n = entrada.read(buf)
                        if (n < 0) break
                        bytes += n
                        if (bytes > LIMITE_BYTES) throw IllegalStateException("lista demasiado grande")
                        salida.write(buf, 0, n)
                    }
                }
            }
            val validas = contarEntradas(tmp)
            if (validas < MIN_ENTRADAS) {
                tmp.delete()
                throw IllegalStateException("no parece una lista válida ($validas entradas)")
            }
            if (!tmp.renameTo(destino)) {
                destino.delete()
                if (!tmp.renameTo(destino)) throw IllegalStateException("no se pudo guardar la lista")
            }
            a.setCabecerasCache(lista.id, conn.getHeaderField("ETag"), conn.getHeaderField("Last-Modified"))
            return true
        } finally {
            conn.disconnect()
        }
    }

    private fun contarEntradas(f: File): Int {
        var n = 0
        f.useLines(Charsets.UTF_8) { lineas ->
            for (l in lineas) {
                if (ParserListas.procesarLinea(l) != null) {
                    n++
                    if (n >= MIN_ENTRADAS) break
                }
            }
        }
        return n
    }

    /** Procesa las listas activas, guarda la caché y cambia el filtro en caliente. Devuelve el nº de dominios. */
    private fun compilar(c: Context): Int {
        val bloqueo = ConjuntoHash64()
        val permitido = ConjuntoHash64()
        for (lista in activas()) {
            val f = archivoLista(c, lista.id)
            if (!f.exists()) continue
            f.useLines(Charsets.UTF_8) { lineas ->
                for (l in lineas) {
                    val e = ParserListas.procesarLinea(l) ?: continue
                    if (e.permitido) permitido.agregar(HashDominio.completo(e.dominio))
                    else bloqueo.agregar(HashDominio.completo(e.dominio))
                }
            }
        }
        val tablas = FiltroListas.Tablas(bloqueo, permitido)
        try {
            filtro.guardarCache(archivoCache(c), tablas)
        } catch (e: Exception) {
            // Sin caché el filtro funciona igual; solo tardará más en arrancar.
        }
        filtro.reemplazar(tablas)
        return bloqueo.tamano
    }
}

    private val c = contexto.applicationContext

    private val reglas: ReglasAnuncios by lazy {
        val f = File(c.filesDir, "reglas_anuncios.json")
        var r: ReglasAnuncios? = null
        if (f.exists()) {
            try { r = ReglasAnuncios.parse(f.readText()) } catch (e: Exception) { r = null }
        }
        r ?: ReglasAnuncios.parse(leerTexto("reglas.json"))
    }

    private val juguete: ByteArray? by lazy {
        try { c.assets.open("anuncios/vacio.mp4").use { it.readBytes() } } catch (e: Exception) { null }
    }

    private val textos = HashMap<String, String>()

    private fun leerTexto(nombre: String): String = synchronized(textos) {
        textos.getOrPut(nombre) {
            try { c.assets.open("anuncios/$nombre").bufferedReader().use { it.readText() } } catch (e: Exception) { "" }
        }
    }

    private val manejadores = WeakHashMap<WebView, MutableList<ScriptHandler>>()
    private val soportaInicio: Boolean by lazy {
        try { WebViewFeature.isFeatureSupported(WebViewFeature.DOCUMENT_START_SCRIPT) } catch (e: Throwable) { false }
    }

    // ---------------------------------------------------------------- red

    /** Respuesta inventada o bloqueo para una petición de subrecurso; null si se deja pasar. */
    fun interceptar(request: WebResourceRequest): WebResourceResponse? {
        if (request.isForMainFrame) return null
        val uri = request.url
        val host = (uri.host ?: return null).lowercase()
        val ruta = (uri.encodedPath ?: "").lowercase()
        val consulta = uri.encodedQuery
        val url = host + ruta + (if (consulta != null) "?" + consulta.lowercase() else "")
        val r = reglas
        val listaNegra = AdBlock.bloquea(host)

        // 1) Anuncio de juguete: .mp4 de dominios de la lista, o URL que encaja con un patrón de «juguete».
        val patronJuguete = r.juguete.firstOrNull { url.contains(it) }
        if (ajustes.tecnica("juguete") && ((ruta.endsWith(".mp4") && listaNegra) || patronJuguete != null)) {
            val resp = respuestaJuguete(request)
            if (resp != null) return resp
        }
        // 2) Scripts sustitutos.
        if (ajustes.tecnica("stubs")) {
            for ((patron, stub) in r.stubs) {
                if (!url.contains(patron)) continue
                val js = when (stub) {
                    "gpt" -> leerTexto("gpt.js")
                    "ima" -> leerTexto("ima.js")
                    "adsbygoogle" -> leerTexto("adsbygoogle.js")
                    else -> ""
                }
                if (js.isEmpty()) continue
                return conCors(request, "application/javascript", js.toByteArray(Charsets.UTF_8))
            }
        }
        // 3) VAST / VMAP vacíos.
        if (ajustes.tecnica("vast") && r.vast.any { url.contains(it) }) {
            val vmap = url.contains("vmap")
            return conCors(request, "application/xml", (if (vmap) VMAP_VACIO else VAST_VACIO).toByteArray(Charsets.UTF_8))
        }
        // 4) Lista «permitir»: se deja pasar (p. ej. el script que detecta bloqueadores y rompería la web).
        if (r.permitir.any { url.contains(it) }) return null
        // 5) Bloqueo clásico por dominio y, opcionalmente, por rutas típicas de anuncios; respuesta vacía con CORS.
        if (listaNegra || (ajustes.tecnica("rutas") && RUTAS.any { ruta.contains(it) })) {
            val mime = if (ruta.endsWith(".js")) "application/javascript" else "text/plain"
            return conCors(request, mime, ByteArray(0))
        }
        return null
    }

    private fun cabecera(request: WebResourceRequest, nombre: String): String? {
        val h = request.requestHeaders ?: return null
        for ((k, v) in h) if (k.equals(nombre, ignoreCase = true)) return v
        return null
    }

    private fun razon(estado: Int): String = when (estado) {
        206 -> "Partial Content"
        416 -> "Range Not Satisfiable"
        else -> "OK"
    }

    /** Respuesta con cabeceras CORS: sin ellas el emisor la toma por fallo y reintenta en bucle. */
    private fun conCors(
        request: WebResourceRequest, mime: String, datos: ByteArray,
        estado: Int = 200, extra: Map<String, String> = emptyMap()
    ): WebResourceResponse {
        val h = HashMap<String, String>()
        val origen = cabecera(request, "Origin")
        h["Access-Control-Allow-Origin"] = origen ?: "*"
        if (origen != null) {
            h["Access-Control-Allow-Credentials"] = "true"
            h["Vary"] = "Origin"
        }
        h["Access-Control-Allow-Methods"] = "GET, POST, HEAD, OPTIONS"
        h["Access-Control-Allow-Headers"] = cabecera(request, "Access-Control-Request-Headers") ?: "*"
        h["Cache-Control"] = "no-store"
        h["Content-Length"] = datos.size.toString()
        h.putAll(extra)
        val codificacion = if (mime.startsWith("text/") || mime.contains("javascript") || mime.contains("xml")) "utf-8" else null
        return WebResourceResponse(mime, codificacion, estado, razon(estado), h, ByteArrayInputStream(datos))
    }

    /** MP4 válido de 0,3 s. Un <video> pide «bytes=0-» y espera un 206, así que se atiende Range (206/416). */
    private fun respuestaJuguete(request: WebResourceRequest): WebResourceResponse? {
        val v = juguete ?: return null
        val n = v.size.toLong()
        val extra = HashMap<String, String>()
        extra["Accept-Ranges"] = "bytes"
        val rango = cabecera(request, "Range")
        val m = if (rango != null) Regex("bytes=(\\d*)-(\\d*)").find(rango) else null
        if (m == null || (m.groupValues[1].isEmpty() && m.groupValues[2].isEmpty())) {
            return conCors(request, "video/mp4", v, 200, extra)
        }
        val a = m.groupValues[1].toLongOrNull()
        val b = m.groupValues[2].toLongOrNull()
        val desde: Long
        val hasta: Long
        if (a == null) {
            val k = (b ?: 0L).coerceAtMost(n)
            desde = n - k
            hasta = n - 1L
        } else {
            desde = a
            hasta = if (b == null) n - 1L else minOf(b, n - 1L)
        }
        if (desde >= n || desde > hasta) {
            extra["Content-Range"] = "bytes */$n"
            return conCors(request, "video/mp4", ByteArray(0), 416, extra)
        }
        extra["Content-Range"] = "bytes $desde-$hasta/$n"
        return conCors(request, "video/mp4", v.copyOfRange(desde.toInt(), hasta.toInt() + 1), 206, extra)
    }

    // ---------------------------------------------------------------- página

    private fun esYoutube(h: String): Boolean =
        h == "youtube.com" || h.endsWith(".youtube.com") || h.endsWith("youtube-nocookie.com")

    /** Script combinado de las técnicas de página [tecnicas] para el sitio [sitio] («*» o un dominio); null si no hay. */
    private fun guion(tecnicas: Set<String>, sitio: String): String? {
        val sb = StringBuilder()
        if (tecnicas.contains("yt") && ajustes.tecnica("yt")) sb.append(envolver(leerTexto("yt_prune.js")))
        if (tecnicas.contains("skip") && ajustes.tecnica("skip")) {
            sb.append(envolver(leerTexto(if (esYoutube(sitio)) "skip_yt.js" else "skip_generico.js")))
        }
        if (tecnicas.contains("css") && ajustes.tecnica("css")) {
            val css = CSS_GENERICO + (if (tecnicas.contains("yt")) "\n" + leerTexto("yt.css").trim() else "")
            sb.append(envolver(CSS_JS.replace("__CSS__", JSONObject.quote(css))))
        }
        if (sb.isEmpty()) return null
        // El indicador evita ejecutar dos veces lo mismo cuando coinciden el script de «*» y el del sitio.
        return "(function(){ if(window.__optimoAds) return; window.__optimoAds=1;\n$sb})();"
    }

    /** Script que oculta lo que el usuario marcó en [host]: una hoja de estilo por selector (uno inválido no estropea el resto). */
    fun guionUsuario(host: String): String? {
        val sels = ReglasUsuario.selectoresPara(host)
        if (sels.isEmpty()) return null
        val lista = JSONArray()
        for (s in sels) lista.put(s)
        return CSS_USUARIO_JS.replace("__LISTA__", lista.toString())
    }

    /** Oculta ya en la página abierta lo marcado para [host] (las visitas futuras lo hacen los scripts de inicio). */
    fun aplicarAhora(web: WebView, host: String) {
        val js = guionUsuario(host) ?: return
        web.evaluateJavascript(js, null)
    }

    /** Script que identifica el anuncio en el punto (x, y) en píxeles CSS; devuelve JSON o null. */
    fun scriptElegir(x: Float, y: Float): String =
        "(" + leerTexto("elegir.js").trim() + ")(" + x + "," + y + ")"

    private fun envolver(js: String): String = if (js.isBlank()) "" else "try{\n$js\n}catch(e){}\n"

    private fun reglasDeOrigen(sitio: String): Set<String> =
        if (sitio == "*") setOf("*")
        else setOf("https://$sitio", "https://*.$sitio", "http://$sitio", "http://*.$sitio")

    /** (Re)instala los scripts de inicio en [web]; llamar al crear la pestaña y al cambiar ajustes. */
    fun instalar(web: WebView) {
        manejadores.remove(web)?.forEach { try { it.remove() } catch (e: Exception) { } }
        if (!ajustes.bloquearAnuncios || !soportaInicio) return
        val lista = ArrayList<ScriptHandler>()
        // Primero los sitios concretos (llevan la unión de «*» y lo suyo) y al final «*»: el indicador evita duplicados.
        val sitios = reglas.sitios.map { it.first }.filter { it != "*" }.distinct()
        for (s in sitios) {
            val js = guion(reglas.tecnicasPara(s), s) ?: continue
            try {
                lista.add(WebViewCompat.addDocumentStartJavaScript(web, js, reglasDeOrigen(s)))
            } catch (e: Exception) { }
        }
        val global = guion(reglas.tecnicasDe("*"), "*")
        if (global != null) {
            try {
                lista.add(WebViewCompat.addDocumentStartJavaScript(web, global, reglasDeOrigen("*")))
            } catch (e: Exception) { }
        }
        // Reglas «ocultar elemento» del usuario, por sitio.
        for (h in ReglasUsuario.hostsConReglas()) {
            val js = guionUsuario(h) ?: continue
            try {
                lista.add(WebViewCompat.addDocumentStartJavaScript(web, js, reglasDeOrigen(h)))
            } catch (e: Exception) { }
        }
        manejadores[web] = lista
    }

    /** Respaldo si el WebView no soporta scripts de inicio: se ejecuta al empezar la página. */
    fun alEmpezar(web: WebView, url: String?) {
        if (soportaInicio || !ajustes.bloquearAnuncios || url == null) return
        val host = try { android.net.Uri.parse(url).host?.lowercase() } catch (e: Exception) { null } ?: return
        guion(reglas.tecnicasPara(host), host)?.let { web.evaluateJavascript(it, null) }
        guionUsuario(host)?.let { web.evaluateJavascript(it, null) }
    }

        )

package com.optimo.navegador

import java.net.URLEncoder

/** Buscador seleccionable. [base] termina en "?q=" (o equivalente) y se le añade la consulta codificada. */
class Motor(val id: String, val nombre: String, val base: String) {
    /** URL del formulario de busqueda (sin la consulta). */
    fun accion(): String = base.substringBefore('?')

    /** Nombre del parametro de la consulta (por defecto "q"). */
    fun parametro(): String {
        val q = base.substringAfter('?', "q=")
        val nombre = q.substringBefore('=')
        return if (nombre.isEmpty()) "q" else nombre
    }
}

/**
 * Resolucion del texto del omnibox (reglas del omnibox de NIA, simplificadas y sin
 * dependencias): vacio; http/https explicito; esquema externo (mailto, tel...);
 * esquema bloqueado; parece dominio (https implicito, http en redes locales); si no, busqueda.
 */
object Omnibox {

    val MOTORES: List<Motor> = listOf(
        Motor("ddg", "DuckDuckGo", "https://duckduckgo.com/?q="),
        Motor("google", "Google", "https://www.google.com/search?q="),
        Motor("bing", "Bing", "https://www.bing.com/search?q="),
        Motor("brave", "Brave Search", "https://search.brave.com/search?q="),
        Motor("ecosia", "Ecosia", "https://www.ecosia.org/search?q=")
    )

    fun motor(id: String): Motor {
        for (m in MOTORES) {
            if (m.id == id) return m
        }
        return MOTORES[0]
    }

    sealed class Resolucion {
        object Vacio : Resolucion()
        class Navegar(val url: String) : Resolucion()
        class Buscar(val url: String) : Resolucion()
        class Externo(val url: String) : Resolucion()
        class NoSoportado(val esquema: String) : Resolucion()
    }

    private val EXTERNOS = setOf("mailto", "tel", "sms", "geo")
    private val RE_ESQUEMA = Regex("^[a-zA-Z][a-zA-Z0-9+.-]*$")
    private val CONOCIDOS = setOf(
        "about", "file", "chrome", "javascript", "data", "view-source", "mailto",
        "ftp", "ws", "wss", "blob", "tel", "sms", "geo", "intent", "content"
    )

    fun resolver(texto: String, motor: Motor): Resolucion {
        val t = texto.trim()
        if (t.isEmpty()) return Resolucion.Vacio
        val esquema = esquemaExplicito(t)
        if (esquema == "http" || esquema == "https") {
            return if (hostDe(t).isNotEmpty()) {
                Resolucion.Navegar(t)
            } else {
                Resolucion.Buscar(motor.base + codificar(t))
            }
        }
        if (esquema != null) {
            return if (esquema in EXTERNOS) Resolucion.Externo(t) else Resolucion.NoSoportado(esquema)
        }
        if (!pareceDominio(t)) return Resolucion.Buscar(motor.base + codificar(t))
        val prefijo = if (esLocal(hostDe(t))) "http://" else "https://"
        return Resolucion.Navegar(prefijo + t)
    }

    /** Esquema explicito en minusculas, o null. "host:puerto" no es un esquema. */
    fun esquemaExplicito(texto: String): String? {
        val i = texto.indexOf(':')
        if (i <= 0) return null
        val candidato = texto.substring(0, i)
        if (!RE_ESQUEMA.matches(candidato)) return null
        val min = candidato.lowercase()
        if (min == "http" || min == "https") return min
        if (texto.substring(i + 1).startsWith("//")) return min
        return if (min in CONOCIDOS) min else null
    }

    /** Host de una URL o de un texto sin esquema ("example.com:80/ruta" -> "example.com"). */
    fun hostDe(texto: String): String {
        var s = texto
        val i = s.indexOf("://")
        if (i >= 0) s = s.substring(i + 3)
        var fin = s.length
        for (c in charArrayOf('/', '?', '#')) {
            val k = s.indexOf(c)
            if (k in 0 until fin) fin = k
        }
        s = s.substring(0, fin)
        val arroba = s.lastIndexOf('@')
        if (arroba >= 0) s = s.substring(arroba + 1)
        if (s.startsWith("[")) {
            val c = s.indexOf(']')
            return if (c > 0) s.substring(0, c + 1) else ""
        }
        val p = s.indexOf(':')
        if (p >= 0) s = s.substring(0, p)
        return s
    }

    /** Sin espacios, sin arroba (correo), localhost, IP o host con punto (no "3.14"). */
    fun pareceDominio(texto: String): Boolean {
        if (texto.any { it <= ' ' }) return false
        val autoridad = texto.substringBefore('/').substringBefore('?').substringBefore('#')
        if (autoridad.contains('@')) return false
        if (autoridad.startsWith("[")) return true
        val host = hostDe(texto)
        if (host.isEmpty()) return false
        if (host == "localhost") return true
        if (!host.contains('.')) return false
        val partes = host.split('.')
        if (partes.any { it.isEmpty() }) return false
        if (partes.all { p -> p.all { it.isDigit() } }) return partes.size == 4
        return true
    }

    /** Hosts a los que se accede por http por defecto: red local (localhost, .local e IP de rangos privados). */
    fun esLocal(host: String): Boolean {
        val h = host.lowercase()
        if (h == "localhost" || h.endsWith(".localhost") || h.endsWith(".local")) return true
        if (h.startsWith("[")) {
            return h == "[::1]" || h.startsWith("[fe80:") || h.startsWith("[fc") || h.startsWith("[fd")
        }
        val partes = h.split('.')
        if (partes.size != 4 || partes.any { p -> p.isEmpty() || !p.all { c -> c.isDigit() } }) return false
        val a = partes[0].toIntOrNull() ?: return false
        val b = partes[1].toIntOrNull() ?: return false
        return a == 10 || a == 127 || (a == 192 && b == 168) || (a == 172 && b in 16..31) || (a == 169 && b == 254)
    }

    fun codificar(q: String): String = URLEncoder.encode(q, "UTF-8")
}

package com.optimo.navegador

import android.content.Context
import org.json.JSONArray
import org.json.JSONObject
import java.io.File

/** Regla «ocultar elemento»: en el sitio [host] (y subdominios) se oculta lo que encaje con [sel]. */
class ReglaOcultar(val host: String, val sel: String)

/**
 * Reglas que añade el usuario al mantener pulsado un anuncio. Se guardan en un JSON pequeño.
 *  - Ocultar elemento: selector CSS por sitio.
 *  - Bloquear dominio: vale para todas las webs y gana a las excepciones de las listas.
 * Las lecturas (desde hilos de red) usan instantáneas inmutables; las escrituras van en el hilo principal.
 */
object ReglasUsuario {

    @Volatile
    private var ocultar: List<ReglaOcultar> = emptyList()

    @Volatile
    private var dominios: Set<String> = emptySet()

    private var archivo: File? = null

    fun iniciar(c: Context) {
        archivo = File(c.applicationContext.filesDir, "reglas_usuario.json")
        cargar()
    }

    val total: Int get() = ocultar.size + dominios.size

    fun listaOcultar(): List<ReglaOcultar> = ocultar

    fun listaDominios(): List<String> = dominios.sorted()

    /** Sitio normalizado: minúsculas y sin «www.» (la regla vale también para sus subdominios). */
    fun clave(host: String): String = host.lowercase().removePrefix("www.")

    fun agregarOcultar(host: String, sel: String): Boolean {
        val h = clave(host)
        if (h.isEmpty() || sel.isBlank()) return false
        if (ocultar.any { it.host == h && it.sel == sel }) return false
        ocultar = ocultar + ReglaOcultar(h, sel)
        guardar()
        return true
    }

    fun agregarDominio(dominio: String): Boolean {
        val d = UtilDominio.normalizar(dominio.removePrefix("www.")) ?: return false
        if (dominios.contains(d)) return false
        dominios = dominios + d
        guardar()
        return true
    }

    fun quitarOcultar(r: ReglaOcultar) {
        ocultar = ocultar.filter { it !== r }
        guardar()
    }

    fun quitarDominio(d: String) {
        dominios = dominios - d
        guardar()
    }

    /** Selectores de ocultación que aplican a la página en [host]. */
    fun selectoresPara(host: String): List<String> {
        val h = host.lowercase()
        return ocultar.filter { h == it.host || h.endsWith("." + it.host) }.map { it.sel }
    }

    fun hostsConReglas(): List<String> = ocultar.map { it.host }.distinct()

    /** True si [host] o alguno de sus dominios padre está en los dominios bloqueados por el usuario. */
    fun bloqueaDominio(host: String): Boolean {
        val set = dominios
        if (set.isEmpty()) return false
        var h = host.lowercase()
        while (true) {
            if (set.contains(h)) return true
            val i = h.indexOf('.')
            if (i < 0) return false
            h = h.substring(i + 1)
            if (h.indexOf('.') < 0) return false
        }
    }

    private fun cargar() {
        try {
            val f = archivo ?: return
            val texto = AlmacenSeguro.leerTexto(f) ?: return
            val j = JSONObject(texto)
            val o = ArrayList<ReglaOcultar>()
            j.optJSONArray("ocultar")?.let {
                for (i in 0 until it.length()) {
                    val x = it.getJSONObject(i)
                    o.add(ReglaOcultar(x.getString("host"), x.getString("sel")))
                }
            }
            val d = HashSet<String>()
            j.optJSONArray("dominios")?.let { for (i in 0 until it.length()) d.add(it.getString(i)) }
            ocultar = o
            dominios = d
        } catch (e: Exception) {
            // Archivo dañado: se empieza sin reglas.
        }
    }

    private fun guardar() {
        try {
            val f = archivo ?: return
            val j = JSONObject()
            val o = JSONArray()
            for (r in ocultar) o.put(JSONObject().put("host", r.host).put("sel", r.sel))
            j.put("ocultar", o)
            val d = JSONArray()
            for (x in dominios.sorted()) d.put(x)
            j.put("dominios", d)
            AlmacenSeguro.escribirTexto(f, j.toString())
        } catch (e: Exception) {
            // Sin espacio o sin permiso: se pierde solo esta escritura.
        }
    }
}

  function prune(o){

      if(o.playerResponse) prune(o.playerResponse);
      if(o.response) prune(o.response);

=====[FUSION:ARCHIVO] _raiz/README.md =====
# Optimo

Navegador Android ligero sobre el WebView del sistema (motor Chromium). Paquete `com.optimo.navegador`,
pensado para compilarse desde Fusion. Una sola dependencia (androidx.webkit); el resto son APIs del sistema.

## Funciones
- Pestañas normales y privadas (la privada usa un perfil de WebView aislado, con cookies, almacenamiento y caché propios que se destruyen al cerrar la última pestaña privada; no guarda historial y bloquea capturas de pantalla).
- Omnibox con busqueda (DuckDuckGo, Google, Bing, Brave, Ecosia) y HTTPS primero con respaldo a http.
- Marcadores, historial, pagina de inicio con accesos directos, sesion restaurada al reabrir.
- Descargas (DownloadManager), buscar en la pagina, compartir, abrir enlaces de otras apps y "compartir con Optimo".
- Bloqueo de anuncios/rastreadores por dominio mas las listas descargables de AntiAds (OISD, HaGeZi, AdAway, EasyList... en `Listas.kt`, con cache compilada) reglas propias (mantener pulsado un anuncio: ocultar elemento por sitio o bloquear dominio; `ReglasUsuario.kt`) y tecnicas extra portadas de AntiAds (`MotorAnuncios.kt`, `assets/anuncios/`): anuncio de juguete (MP4 valido), VAST vacio, scripts sustitutos GPT/IMA/AdSense, CSS de huecos, YouTube (datos de anuncios y saltador); se activan una a una en Ajustes, versión de escritorio, modo oscuro de paginas (experimental).
- Permisos de camara, microfono y ubicacion con confirmacion; subida de archivos; pantalla completa de video.
- Certificados no válidos: bloqueados sin opción de saltarlos (solo equipos de la red local, y tras confirmar).
- Sitios sin HTTPS: nunca se abren por http sin confirmación; descargas sin cifrar bloqueadas; contenido mixto bloqueado.
- Datos locales cifrados (marcadores, historial, sesión y reglas) con AES-256-GCM y clave en el Keystore (`AlmacenSeguro.kt`).
- Botón ⛶ en el extremo derecho de la barra de direcciones: página a pantalla completa (sin barras de Optimo ni del sistema); se sale pulsando atrás dos veces.
- Compatible con el autorrelleno de La Bóveda (selecciónala como servicio de autorrelleno de Android).

## Firma
El workflow de Fusion genera `apk-build/debug.keystore` la primera vez que compila en el repositorio de este
proyecto y lo conserva: firma propia, distinta de cualquier otra app, estable entre builds.

## Estructura
`app/src/main/java/com/optimo/navegador/`: `MainActivity.kt` (interfaz, pestañas, menus, clientes WebView),
`Omnibox.kt`, `AdBlock.kt`, `Paginas.kt` (logica pura), `Almacen.kt` (marcadores, historial, ajustes), `AlmacenSeguro.kt` (cifrado local), `Modelo.kt`.
